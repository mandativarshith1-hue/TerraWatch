import { createContext, useContext, useEffect, useState, type ReactNode } from "react";
import type { ThemeMode } from "@/types";

interface ThemeContextValue {
  mode: ThemeMode;
  toggle: () => void;
  setMode: (mode: ThemeMode) => void;
}

const ThemeContext = createContext<ThemeContextValue | null>(null);

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [mode, setMode] = useState<ThemeMode>(() => {
    const saved = localStorage.getItem("terrwatch-theme");
    return (saved as ThemeMode) ?? "dark";
  });

  useEffect(() => {
    localStorage.setItem("terrwatch-theme", mode);
    document.documentElement.classList.toggle("light-mode", mode === "light");
  }, [mode]);

  const toggle = () => setMode((m) => (m === "dark" ? "light" : "dark"));

  return <ThemeContext.Provider value={{ mode, toggle, setMode }}>{children}</ThemeContext.Provider>;
}

export function useTheme() {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error("useTheme must be used within ThemeProvider");
  return ctx;
}


//=== FILE: src/context/LocationContext.tsx ===
import { createContext, useContext, useState, useCallback, type ReactNode } from "react";
import type { LatLon, CityLocation } from "@/types";

interface LocationContextValue {
  selectedLocation: (CityLocation & { isCustom?: boolean }) | null;
  setSelectedLocation: (loc: (CityLocation & { isCustom?: boolean }) | null) => void;
  flyTo: LatLon | null;
  setFlyTo: (latlon: LatLon | null) => void;
  watchlist: CityLocation[];
  toggleWatchlist: (city: CityLocation) => void;
  isWatched: (id: string) => boolean;
}

const LocationContext = createContext<LocationContextValue | null>(null);

export function LocationProvider({ children }: { children: ReactNode }) {
  const [selectedLocation, setSelectedLocation] = useState<(CityLocation & { isCustom?: boolean }) | null>(null);
  const [flyTo, setFlyTo] = useState<LatLon | null>(null);
  const [watchlist, setWatchlist] = useState<CityLocation[]>(() => {
    try {
      return JSON.parse(localStorage.getItem("terrwatch-watchlist") ?? "[]");
    } catch {
      return [];
    }
  });

  const toggleWatchlist = useCallback((city: CityLocation) => {
    setWatchlist((prev) => {
      const exists = prev.some((c) => c.id === city.id);
      const next = exists ? prev.filter((c) => c.id !== city.id) : [...prev, city];
      localStorage.setItem("terrwatch-watchlist", JSON.stringify(next));
      return next;
    });
  }, []);

  const isWatched = useCallback((id: string) => watchlist.some((c) => c.id === id), [watchlist]);

  return (
    <LocationContext.Provider
      value={{ selectedLocation, setSelectedLocation, flyTo, setFlyTo, watchlist, toggleWatchlist, isWatched }}
    >
      {children}
    </LocationContext.Provider>
  );
}

export function useLocation() {
  const ctx = useContext(LocationContext);
  if (!ctx) throw new Error("useLocation must be used within LocationProvider");
  return ctx;
}

function makeBaseWeather(overrides: Partial<WeatherData>): WeatherData {
  const now = new Date();
  const hourTimes: string[] = [];
  const dayTimes: string[] = [];
  for (let i = 0; i < 168; i++) {
    const d = new Date(now);
    d.setHours(d.getHours() + i);
    hourTimes.push(d.toISOString());
  }
  for (let i = 0; i < 7; i++) {
    const d = new Date(now);
    d.setDate(d.getDate() + i);
    dayTimes.push(d.toISOString().slice(0, 10));
  }

  return {
    current: {
      temperature: 28,
      apparentTemperature: 30,
      precipitation: 0,
      rain: 0,
      windSpeed: 15,
      windGusts: 25,
      pressure: 1010,
      humidity: 70,
      visibility: 10000,
      cloudCover: 40,
      weatherCode: 3,
      isDay: true,
      time: now.toISOString(),
      ...overrides.current,
    },
    hourly: {
      time: hourTimes,
      temperature: Array(168).fill(28),
      precipitation: Array(168).fill(0),
      rain: Array(168).fill(0),
      windSpeed: Array(168).fill(15),
      windGusts: Array(168).fill(25),
      visibility: Array(168).fill(10000),
      pressure: Array(168).fill(1010),
      humidity: Array(168).fill(70),
      cloudCover: Array(168).fill(40),
      weatherCode: Array(168).fill(3),
      ...overrides.hourly,
    },
    daily: {
      time: dayTimes,
      temperatureMax: [32, 33, 34, 33, 32, 31, 30],
      temperatureMin: [24, 25, 25, 24, 24, 23, 23],
      precipitationSum: [5, 2, 1, 3, 8, 12, 6],
      rainSum: [5, 2, 1, 3, 8, 12, 6],
      windSpeedMax: [25, 20, 18, 22, 30, 28, 24],
      windGustsMax: [35, 30, 25, 32, 45, 40, 35],
      weatherCode: [3, 3, 2, 3, 61, 63, 3],
      sunrise: dayTimes.map((d) => `${d}T06:00`),
      sunset: dayTimes.map((d) => `${d}T18:30`),
      uvIndexMax: [7, 8, 8, 7, 5, 4, 6],
      ...overrides.daily,
    },
    timezone: "UTC",
    fetchedAt: Date.now(),
  };
}

const cache = new Map<string, { data: unknown; expires: number }>();
const CACHE_TTL = 10 * 60 * 1000;

function getCached<T>(key: string): T | null {
  const entry = cache.get(key);
  if (entry && entry.expires > Date.now()) return entry.data as T;
  cache.delete(key);
  return null;
}

function setCached(key: string, data: unknown) {
  cache.set(key, { data, expires: Date.now() + CACHE_TTL });
}

async function fetchJSON<T>(url: string, retries = 2): Promise<T> {
  const cached = getCached<T>(url);
  if (cached) return cached;
  for (let attempt = 0; attempt <= retries; attempt++) {
    try {
      const res = await fetch(url, { signal: AbortSignal.timeout(15000) });
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const data = (await res.json()) as T;
      setCached(url, data);
      return data;
    } catch (err) {
      if (attempt === retries) throw err;
      await new Promise((r) => setTimeout(r, 500 * (attempt + 1)));
    }
  }
  throw new Error("Unreachable");
}

export async function fetchHistorical(lat: number, lon: number): Promise<HistoricalData> {
  const now = new Date();
  const end = new Date(now);
  end.setDate(end.getDate() - 1);
  const start = new Date(end);
  start.setDate(start.getDate() - 30);

  const params = new URLSearchParams({
    latitude: lat.toFixed(4),
    longitude: lon.toFixed(4),
    start_date: start.toISOString().slice(0, 10),
    end_date: end.toISOString().slice(0, 10),
    daily: "rain_sum,temperature_2m_mean",
    timezone: "auto",
  });

  try {
    const data = await fetchJSON<Record<string, unknown>>(`${OPEN_METEO_ARCHIVE}?${params}`);
    const daily = data.daily as Record<string, number[] | string[]>;
    return {
      dates: daily.time as string[],
      rainfall: daily.rain_sum as number[],
      temperature: daily.temperature_2m_mean as number[],
      source: "Open-Meteo Archive API",
    };
  } catch {
    return { dates: [], rainfall: [], temperature: [], source: "Open-Meteo Archive API" };
  }
}

export async function fetchFloodData(lat: number, lon: number): Promise<number | null> {
  const params = new URLSearchParams({
    latitude: lat.toFixed(4),
    longitude: lon.toFixed(4),
    daily: "river_discharge",
    past_days: "7",
  });
  try {
    const data = await fetchJSON<Record<string, unknown>>(`${OPEN_METEO_FLOOD}?${params}`);
    const daily = data.daily as Record<string, number[]>;
    const discharge = (daily.river_discharge as number[]) ?? [];
    return discharge.length > 0 ? discharge[discharge.length - 1] : null;
  } catch {
    return null;
  }
}

interface USGSFeature {
  id: string;
  properties: {
    mag: number;
    place: string;
    time: number;
    tsunami: number;
    type: string;
  };
  geometry: { coordinates: [number, number, number] };
}

export async function fetchEarthquakes(): Promise<DisasterEvent[]> {
  try {
    const [dayData, weekData] = await Promise.all([
      fetchJSON<{ features: USGSFeature[] }>(USGS_EARTHQUAKE_DAY).catch(() => ({ features: [] })),
      fetchJSON<{ features: USGSFeature[] }>(USGS_EARTHQUAKE_WEEK).catch(() => ({ features: [] })),
    ]);

    const seen = new Set<string>();
    const events: DisasterEvent[] = [];

    for (const feature of [...dayData.features, ...weekData.features]) {
      if (seen.has(feature.id)) continue;
      seen.add(feature.id);
      const [lon, lat, depth] = feature.geometry.coordinates;
      const mag = feature.properties.mag ?? 0;
      const severity: AlertSeverity = mag >= 6 ? "emergency" : mag >= 4.5 ? "warning" : mag >= 2.5 ? "watch" : "advisory";
      const riskLevel: RiskLevel = mag >= 6 ? "severe" : mag >= 4.5 ? "high" : mag >= 2.5 ? "moderate" : "low";

      events.push({
        id: `usgs-${feature.id}`,
        type: "earthquake",
        title: `M${mag.toFixed(1)} Earthquake`,
        description: `${feature.properties.place ?? "Unknown location"}`,
        location: feature.properties.place ?? "Unknown",
        lat,
        lon,
        severity,
        source: "usgs",
        sourceUrl: `https://earthquake.usgs.gov/earthquakes/eventpage/${feature.id}`,
        time: new Date(feature.properties.time).toISOString(),
        magnitude: mag,
        depth,
        tsunamiWarning: feature.properties.tsunami === 1,
        riskLevel,
        raw: feature,
      });
    }
    return events;
  } catch {
    return [];
  }
}

interface EONETEvent {
  id: string;
  title: string;
  description?: string;
  categories: { id: string; title: string }[];
  geometry: { date: string; type: string; coordinates: number[] | number[][] }[];
  sources: { url: string; id: string }[];
  closed?: string | null;
}

const EONET_CATEGORY_MAP: Record<string, HazardType> = {
  "wildfires": "wildfire",
  "volcanoes": "volcano",
  "severeStorms": "cyclone",
  "floods": "flood",
  "drought": "drought",
  "landslides": "landslide",
  "seaLakeIce": "coldWave",
  "manmade": "wildfire",
  "tempExtremes": "heatwave",
};

export async function fetchEonetEvents(): Promise<DisasterEvent[]> {
  try {
    const data = await fetchJSON<{ events: EONETEvent[] }>(`${EONET_EVENTS}?status=open`);
    const events: DisasterEvent[] = [];

    for (const event of data.events) {
      const category = event.categories[0]?.id ?? "";
      const hazardType = EONET_CATEGORY_MAP[category] ?? "flood";
      const geo = event.geometry[event.geometry.length - 1];
      if (!geo || geo.type === "Polygon") continue;
      const coords = geo.coordinates as number[];
      const [lon, lat] = coords;
      if (lat === undefined || lon === undefined) continue;

      const riskLevel: RiskLevel = "high";
      const severity: AlertSeverity = "warning";

      events.push({
        id: `eonet-${event.id}`,
        type: hazardType,
        title: event.title,
        description: event.description ?? event.categories.map((c) => c.title).join(", "),
        location: `${lat.toFixed(2)}, ${lon.toFixed(2)}`,
        lat,
        lon,
        severity,
        source: "eonet",
        sourceUrl: event.sources[0]?.url,
        time: geo.date,
        riskLevel,
        raw: event,
      });
    }
    return events;
  } catch {
    return [];
  }
}

interface NWSAlert {
  id: string;
  event: string;
  headline: string;
  description: string;
  instruction: string;
  severity: string;
  areaDesc: string;
  onset: string;
  expires: string;
  sent: string;
  geometry?: { coordinates: number[][] | number[][][] };
  parameters?: Record<string, string[]>;
}

export async function fetchNWSAlerts(): Promise<UnifiedAlert[]> {
  try {
    const data = await fetchJSON<{ features: { id: string; properties: NWSAlert; geometry?: { coordinates: number[][] } }[] }>(
      `${NWS_ALERTS}?status=actual&message_type=alert`
    );
    const alerts: UnifiedAlert[] = [];

    for (const feature of data.features) {
      const p = feature.properties;
      if (!p) continue;
      const severityMap: Record<string, AlertSeverity> = {
        Minor: "advisory",
        Moderate: "watch",
        Severe: "warning",
        Extreme: "emergency",
      };
      const riskMap: Record<string, RiskLevel> = {
        Minor: "low",
        Moderate: "moderate",
        Severe: "high",
        Extreme: "severe",
      };
      const sev = severityMap[p.severity] ?? "watch";
      const risk = riskMap[p.severity] ?? "moderate";

      let lat = 39.5,
        lon = -98.35;
      if (feature.geometry?.coordinates) {
        const coords = feature.geometry.coordinates[0] as unknown;
        if (Array.isArray(coords) && Array.isArray((coords as unknown[])[0])) {
          const polyCoords = coords as number[][];
          lon = polyCoords.reduce((s, c) => s + c[0], 0) / polyCoords.length;
          lat = polyCoords.reduce((s, c) => s + c[1], 0) / polyCoords.length;
        }
      }

      const titleLower = (p.event || "").toLowerCase();
      let type: HazardType | "official" = "official";
      if (titleLower.includes("flood")) type = "flood";
      else if (titleLower.includes("heat")) type = "heatwave";
      else if (titleLower.includes("tornado") || titleLower.includes("thunderstorm") || titleLower.includes("wind") || titleLower.includes("hurricane") || titleLower.includes("cyclone")) type = "cyclone";
      else if (titleLower.includes("fire")) type = "wildfire";
      else if (titleLower.includes("winter") || titleLower.includes("cold") || titleLower.includes("snow") || titleLower.includes("ice")) type = "coldWave";
      else if (titleLower.includes("tsunami")) type = "tsunami";

      alerts.push({
        id: `nws-${feature.id}`,
        type,
        title: p.event,
        description: (p.description ?? "").slice(0, 500),
        severity: sev,
        riskLevel: risk,
        source: "nws",
        sourceLabel: "NWS (US National Weather Service)",
        sourceUrl: `https://api.weather.gov/alerts/${feature.id}`,
        time: p.sent,
        lat,
        lon,
        location: p.areaDesc ?? "United States",
        instructions: p.instruction ?? undefined,
        expiry: p.expires,
      });
    }
    return alerts;
  } catch {
    return [];
  }
}

export function deriveWeatherAlerts(
  weather: WeatherData,
  lat: number,
  lon: number,
  locationName: string
): UnifiedAlert[] {
  const alerts: UnifiedAlert[] = [];
  const { current, daily, hourly } = weather;

  const rain24h = daily.rainSum[0] ?? 0;
  const rain72h = (daily.rainSum[0] ?? 0) + (daily.rainSum[1] ?? 0) + (daily.rainSum[2] ?? 0);
  const maxGusts = daily.windGustsMax[0] ?? 0;
  const tempMax = daily.temperatureMax[0] ?? current.temperature;
  const apparentTemp = current.apparentTemperature;
  const visibility = current.visibility;

  if (rain24h > 115.6) {
    alerts.push({
      id: `derived-flood-${lat}-${lon}-${Date.now()}`,
      type: "flood",
      title: "RED Heavy Rainfall Warning",
      description: `24h rainfall: ${rain24h.toFixed(1)}mm exceeds 115.6mm threshold. Extreme flood risk.`,
      severity: "emergency",
      riskLevel: "severe",
      source: "derived",
      sourceLabel: "Derived from forecast (not an official warning)",
      sourceUrl: "https://mausam.imd.gov.in/",
      time: new Date().toISOString(),
      lat,
      lon,
      location: locationName,
    });
  } else if (rain24h > 64.5) {
    alerts.push({
      id: `derived-flood-${lat}-${lon}-${Date.now()}`,
      type: "flood",
      title: "ORANGE Heavy Rainfall Warning",
      description: `24h rainfall: ${rain24h.toFixed(1)}mm exceeds 64.5mm threshold. High flood risk.`,
      severity: "warning",
      riskLevel: "high",
      source: "derived",
      sourceLabel: "Derived from forecast (not an official warning)",
      sourceUrl: "https://mausam.imd.gov.in/",
      time: new Date().toISOString(),
      lat,
      lon,
      location: locationName,
    });
  }

  if (tempMax >= 40 || apparentTemp >= 45) {
    alerts.push({
      id: `derived-heat-${lat}-${lon}-${Date.now()}`,
      type: "heatwave",
      title: tempMax >= 45 ? "RED Heatwave Warning" : "ORANGE Heatwave Warning",
      description: `Max temperature: ${tempMax.toFixed(1)}°C, apparent temp: ${apparentTemp.toFixed(1)}°C. Heatwave conditions.`,
      severity: tempMax >= 45 ? "emergency" : "warning",
      riskLevel: tempMax >= 45 ? "severe" : "high",
      source: "derived",
      sourceLabel: "Derived from forecast (not an official warning)",
      sourceUrl: "https://mausam.imd.gov.in/",
      time: new Date().toISOString(),
      lat,
      lon,
      location: locationName,
    });
  }

  if (maxGusts >= 90) {
    alerts.push({
      id: `derived-wind-${lat}-${lon}-${Date.now()}`,
      type: "cyclone",
      title: "RED Strong Wind Warning",
      description: `Wind gusts up to ${maxGusts.toFixed(0)} km/h expected. Damaging winds likely.`,
      severity: "emergency",
      riskLevel: "severe",
      source: "derived",
      sourceLabel: "Derived from forecast (not an official warning)",
      sourceUrl: "https://mausam.imd.gov.in/",
      time: new Date().toISOString(),
      lat,
      lon,
      location: locationName,
    });
  } else if (maxGusts >= 60) {
    alerts.push({
      id: `derived-wind-${lat}-${lon}-${Date.now()}`,
      type: "cyclone",
      title: "ORANGE Strong Wind Warning",
      description: `Wind gusts up to ${maxGusts.toFixed(0)} km/h expected. Strong winds likely.`,
      severity: "warning",
      riskLevel: "high",
      source: "derived",
      sourceLabel: "Derived from forecast (not an official warning)",
      sourceUrl: "https://mausam.imd.gov.in/",
      time: new Date().toISOString(),
      lat,
      lon,
      location: locationName,
    });
  }

  if (visibility > 0 && visibility < 1000) {
    alerts.push({
      id: `derived-fog-${lat}-${lon}-${Date.now()}`,
      type: "official",
      title: "Dense Fog Advisory",
      description: `Visibility: ${visibility.toFixed(0)}m. Very low visibility conditions.`,
      severity: "watch",
      riskLevel: "moderate",
      source: "derived",
      sourceLabel: "Derived from forecast (not an official warning)",
      sourceUrl: "https://mausam.imd.gov.in/",
      time: new Date().toISOString(),
      lat,
      lon,
      location: locationName,
    });
  }

  const hourlyRain = hourly.rain.slice(0, 24);
  const maxHourlyRain = Math.max(...hourlyRain, 0);
  if (maxHourlyRain > 20) {
    alerts.push({
      id: `derived-extrain-${lat}-${lon}-${Date.now()}`,
      type: "extremeRainfall",
      title: "Extreme Rainfall Warning",
      description: `Hourly rainfall up to ${maxHourlyRain.toFixed(1)}mm. Flash flood risk possible.`,
      severity: "warning",
      riskLevel: "high",
      source: "derived",
      sourceLabel: "Derived from forecast (not an official warning)",
      sourceUrl: "https://mausam.imd.gov.in/",
      time: new Date().toISOString(),
      lat,
      lon,
      location: locationName,
    });
  }

  return alerts;
}

export function haversineDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371;
  const dLat = ((lat2 - lat1) * Math.PI) / 180;
  const dLon = ((lon2 - lon1) * Math.PI) / 180;
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos((lat1 * Math.PI) / 180) * Math.cos((lat2 * Math.PI) / 180) *
    Math.sin(dLon / 2) * Math.sin(dLon / 2);
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

export function scoreToLevel(score: number): RiskLevel {
  if (score >= 70) return "severe";
  if (score >= 45) return "high";
  if (score >= 25) return "moderate";
  return "low";
}

export function levelToScore(level: RiskLevel): number {
  switch (level) {
    case "severe": return 80;
    case "high": return 55;
    case "moderate": return 35;
    case "low": return 10;
  }
}

function clampScore(n: number): number {
  return Math.max(0, Math.min(100, Math.round(n)));
}

function makeIndicator(
  name: string,
  value: string,
  contribution: number,
  maxContribution: number,
  description: string
): RiskIndicator {
  return { name, value, contribution: Math.round(contribution), maxContribution, description };
}

export function assessFlood(weather: WeatherData, riverDischarge: number | null): RiskAssessment {
  const rain24h = weather.daily.rainSum[0] ?? 0;
  const rain72h = (weather.daily.rainSum[0] ?? 0) + (weather.daily.rainSum[1] ?? 0) + (weather.daily.rainSum[2] ?? 0);
  const forecastRain = weather.daily.precipitationSum.slice(0, 3).reduce((a, b) => a + (b ?? 0), 0);
  const humidity = weather.current.humidity;
  const indicators: RiskIndicator[] = [];

  const rain24Score = Math.min(30, (rain24h / 115.6) * 30);
  indicators.push(makeIndicator("24h Rainfall", `${rain24h.toFixed(1)} mm`, rain24Score, 30, "IMD: >64.5mm is heavy rain, >115.6mm is extremely heavy"));

  const rain72Score = Math.min(25, (rain72h / 200) * 25);
  indicators.push(makeIndicator("72h Rainfall", `${rain72h.toFixed(1)} mm`, rain72Score, 25, "3-day accumulation indicates sustained rainfall"));

  const forecastScore = Math.min(20, (forecastRain / 150) * 20);
  indicators.push(makeIndicator("Forecast Rain (3d)", `${forecastRain.toFixed(1)} mm`, forecastScore, 20, "Upcoming precipitation forecast"));

  const dischargeScore = riverDischarge !== null ? Math.min(15, (riverDischarge / 5000) * 15) : 0;
  indicators.push(makeIndicator("River Discharge", riverDischarge !== null ? `${riverDischarge.toFixed(0)} m³/s` : "N/A", dischargeScore, 15, "River flow rate from Open-Meteo Flood API"));

  const humidityScore = Math.min(10, ((humidity - 60) / 40) * 10);
  indicators.push(makeIndicator("Humidity", `${humidity.toFixed(0)}%`, Math.max(0, humidityScore), 10, "High humidity indicates saturated air mass"));

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "flood", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export function assessHeatwave(weather: WeatherData, historical: HistoricalData | null): RiskAssessment {
  const tempMax = weather.daily.temperatureMax[0] ?? weather.current.temperature;
  const apparentTemp = weather.current.apparentTemperature;
  const histAvg = historical && historical.temperature.length > 0
    ? historical.temperature.reduce((a, b) => a + b, 0) / historical.temperature.length
    : tempMax - 5;
  const deviation = tempMax - histAvg;

  let consecutiveHotDays = 0;
  for (const t of weather.daily.temperatureMax) {
    if (t >= 40) consecutiveHotDays++;
    else break;
  }

  const indicators: RiskIndicator[] = [];

  const tempScore = Math.min(35, ((tempMax - 30) / 20) * 35);
  indicators.push(makeIndicator("Max Temperature", `${tempMax.toFixed(1)}°C`, tempScore, 35, "IMD: ≥40°C is heatwave condition"));

  const apparentScore = Math.min(25, ((apparentTemp - 35) / 15) * 25);
  indicators.push(makeIndicator("Apparent Temperature", `${apparentTemp.toFixed(1)}°C`, Math.max(0, apparentScore), 25, "Feels-like temperature including humidity"));

  const devScore = Math.min(20, (Math.max(0, deviation) / 7) * 20);
  indicators.push(makeIndicator("Deviation from Normal", `+${deviation.toFixed(1)}°C`, devScore, 20, "IMD: 4.5°C above normal = heatwave"));

  const consecScore = Math.min(20, (consecutiveHotDays / 5) * 20);
  indicators.push(makeIndicator("Consecutive Hot Days", `${consecutiveHotDays} days`, consecScore, 20, "Sustained heat compounds health risk"));

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "heatwave", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export function assessCyclone(weather: WeatherData): RiskAssessment {
  const windSpeed = weather.current.windSpeed;
  const gusts = weather.current.windGusts;
  const pressure = weather.current.pressure;
  const maxGusts = weather.daily.windGustsMax[0] ?? gusts;

  const indicators: RiskIndicator[] = [];

  const windScore = Math.min(30, (windSpeed / 120) * 30);
  indicators.push(makeIndicator("Wind Speed", `${windSpeed.toFixed(0)} km/h`, windScore, 30, "Sustained wind speed"));

  const gustScore = Math.min(30, (gusts / 150) * 30);
  indicators.push(makeIndicator("Wind Gusts", `${gusts.toFixed(0)} km/h`, gustScore, 30, "IMD: ≥60 km/h orange, ≥90 km/h red"));

  const pressureScore = pressure < 1000 ? Math.min(25, ((1013 - pressure) / 40) * 25) : 5;
  indicators.push(makeIndicator("Pressure", `${pressure.toFixed(0)} hPa`, pressureScore, 25, "Low pressure indicates cyclone/storm system"));

  const forecastGustScore = Math.min(15, (maxGusts / 150) * 15);
  indicators.push(makeIndicator("Forecast Max Gusts", `${maxGusts.toFixed(0)} km/h`, forecastGustScore, 15, "Maximum forecast gusts for today"));

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "cyclone", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export function assessExtremeRainfall(weather: WeatherData): RiskAssessment {

  const maxHourly = Math.max(...hourlyRain, 0);
  const rain3day = weather.daily.rainSum.slice(0, 3).reduce((a, b) => a + (b ?? 0), 0);

  indicators.push(makeIndicator("Max Hourly Rain", `${maxHourly.toFixed(1)} mm/hr`, hourlyScore, 40, ">20mm/hr is intense rainfall"));

  indicators.push(makeIndicator("Current Rain Rate", `${intensity.toFixed(1)} mm`, intensityScore, 25, "Current precipitation intensity"));

  indicators.push(makeIndicator("3-Day Accumulation", `${rain3day.toFixed(1)} mm`, accum3dScore, 20, "Cumulative rainfall over 3 days"));

  const precipScore = Math.min(15, ((weather.daily.precipitationSum[0] ?? 0) / 200) * 15);
  indicators.push(makeIndicator("24h Precipitation", `${(weather.daily.precipitationSum[0] ?? 0).toFixed(1)} mm`, precipScore, 15, "Total precipitation today"));

  return { hazard: "extremeRainfall", score: clampScore(total), level: scoreToLevel(total), indicators };

export function assessEarthquake(lat: number, lon: number, events: DisasterEvent[]): RiskAssessment {
  let nearestDist = Infinity;
  let nearestMag = 0;
  let nearestEvent: DisasterEvent | null = null;

  for (const e of events) {
    if (e.type !== "earthquake") continue;
    const dist = haversineDistance(lat, lon, e.lat, e.lon);
    if (dist < nearestDist) {
      nearestDist = dist;
      nearestMag = e.magnitude ?? 0;
      nearestEvent = e;
    }
  }

  const indicators: RiskIndicator[] = [];

  if (nearestEvent) {
    const proxScore = Math.max(0, 40 - (nearestDist / 500) * 40);
    indicators.push(makeIndicator("Distance to Nearest Quake", `${nearestDist.toFixed(0)} km`, proxScore, 40, "Closer events have higher impact"));

    const magScore = Math.min(40, (nearestMag / 8) * 40);
    indicators.push(makeIndicator("Magnitude", `M${nearestMag.toFixed(1)}`, magScore, 40, "Higher magnitude = greater energy released"));

    const depthScore = nearestEvent.depth
      ? Math.min(20, (50 / Math.max(nearestEvent.depth, 5)) * 20)
      : 10;
    indicators.push(makeIndicator("Depth", nearestEvent.depth ? `${nearestEvent.depth.toFixed(0)} km` : "N/A", depthScore, 20, "Shallow quakes cause more surface damage"));
  } else {
    indicators.push(makeIndicator("Nearby Earthquakes", "None detected (24h)", 5, 40, "No recent seismic activity in vicinity"));
    indicators.push(makeIndicator("Magnitude", "N/A", 0, 40, "No nearby event"));
    indicators.push(makeIndicator("Depth", "N/A", 0, 20, "No nearby event"));
  }

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "earthquake", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export function assessWildfire(lat: number, lon: number, events: DisasterEvent[], weather: WeatherData): RiskAssessment {
  let nearestDist = Infinity;
  let nearestEvent: DisasterEvent | null = null;

  for (const e of events) {
    if (e.type !== "wildfire") continue;
    const dist = haversineDistance(lat, lon, e.lat, e.lon);
    if (dist < nearestDist) {
      nearestDist = dist;
      nearestEvent = e;
    }
  }

  const indicators: RiskIndicator[] = [];
  const temp = weather.current.temperature;
  const humidity = weather.current.humidity;

  const proxScore = nearestEvent ? Math.max(0, 30 - (nearestDist / 200) * 30) : 0;
  indicators.push(makeIndicator("Distance to Active Fire", nearestEvent ? `${nearestDist.toFixed(0)} km` : "None detected", proxScore, 30, "Proximity to active wildfire from NASA EONET"));

  const tempScore = Math.min(25, ((temp - 25) / 20) * 25);
  indicators.push(makeIndicator("Temperature", `${temp.toFixed(1)}°C`, Math.max(0, tempScore), 25, "High temperatures increase fire risk"));

  const humidScore = Math.min(25, ((40 - humidity) / 40) * 25);
  indicators.push(makeIndicator("Humidity", `${humidity.toFixed(0)}%`, Math.max(0, humidScore), 25, "Low humidity dries vegetation, increasing fire risk"));

  const windScore = Math.min(20, (weather.current.windSpeed / 50) * 20);
  indicators.push(makeIndicator("Wind Speed", `${weather.current.windSpeed.toFixed(0)} km/h`, windScore, 20, "Wind spreads fire rapidly"));

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "wildfire", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export function assessLandslide(weather: WeatherData): RiskAssessment {
  const rain24h = weather.daily.rainSum[0] ?? 0;
  const rain72h = (weather.daily.rainSum[0] ?? 0) + (weather.daily.rainSum[1] ?? 0) + (weather.daily.rainSum[2] ?? 0);
  const humidity = weather.current.humidity;

  const indicators: RiskIndicator[] = [];

  const rainScore = Math.min(40, (rain24h / 100) * 40);
  indicators.push(makeIndicator("24h Rainfall", `${rain24h.toFixed(1)} mm`, rainScore, 40, "Heavy rainfall triggers landslides on slopes"));

  const accumScore = Math.min(30, (rain72h / 200) * 30);
  indicators.push(makeIndicator("72h Accumulation", `${rain72h.toFixed(1)} mm`, accumScore, 30, "Sustained rainfall saturates soil"));

  const humidScore = Math.min(30, ((humidity - 60) / 40) * 30);
  indicators.push(makeIndicator("Soil Moisture (proxy)", `${humidity.toFixed(0)}%`, Math.max(0, humidScore), 30, "High humidity indicates saturated soil"));

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "landslide", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export function assessDrought(weather: WeatherData, historical: HistoricalData | null): RiskAssessment {
  const rain7day = weather.daily.rainSum.reduce((a, b) => a + (b ?? 0), 0);
  const histAvgRain = historical && historical.rainfall.length > 0
    ? historical.rainfall.reduce((a, b) => a + b, 0) / historical.rainfall.length
    : 5;
  const tempAvg = weather.daily.temperatureMax.reduce((a, b) => a + (b ?? 0), 0) / (weather.daily.temperatureMax.length || 1);

  const indicators: RiskIndicator[] = [];

  const rainDeficit = Math.max(0, 30 - (rain7day / 50) * 30);
  indicators.push(makeIndicator("7-Day Rainfall", `${rain7day.toFixed(1)} mm`, rainDeficit, 30, "Low rainfall indicates dry conditions"));

  const histScore = Math.max(0, 25 - (histAvgRain / 10) * 25);
  indicators.push(makeIndicator("Historical Avg Rain", `${histAvgRain.toFixed(1)} mm/day`, histScore, 25, "Below-normal rainfall over 30 days"));

  const tempScore = Math.min(25, ((tempAvg - 30) / 15) * 25);
  indicators.push(makeIndicator("Avg Max Temperature", `${tempAvg.toFixed(1)}°C`, Math.max(0, tempScore), 25, "High temperatures increase evaporation"));

  const dryDays = weather.daily.rainSum.filter((r) => (r ?? 0) < 1).length;
  const dryScore = Math.min(20, (dryDays / 7) * 20);
  indicators.push(makeIndicator("Dry Days (7d)", `${dryDays} days`, dryScore, 20, "Consecutive days without rain"));

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "drought", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export function assessColdWave(weather: WeatherData): RiskAssessment {
  const tempMin = weather.daily.temperatureMin[0] ?? weather.current.temperature;
  const tempCurrent = weather.current.temperature;
  const windChill = tempCurrent - (weather.current.windSpeed / 10) * 0.5;

  const indicators: RiskIndicator[] = [];

  const minScore = Math.max(0, ((10 - tempMin) / 15) * 40);
  indicators.push(makeIndicator("Min Temperature", `${tempMin.toFixed(1)}°C`, minScore, 40, "IMD: ≤10°C with wind is cold wave territory"));

  const currentScore = Math.max(0, ((10 - tempCurrent) / 15) * 30);
  indicators.push(makeIndicator("Current Temperature", `${tempCurrent.toFixed(1)}°C`, currentScore, 30, "Current cold exposure"));

  const chillScore = Math.max(0, ((10 - windChill) / 15) * 30);
  indicators.push(makeIndicator("Wind Chill", `${windChill.toFixed(1)}°C`, chillScore, 30, "Wind makes it feel colder than actual temperature"));

  const total = indicators.reduce((s, i) => s + i.contribution, 0);
  return { hazard: "coldWave", score: clampScore(total), level: scoreToLevel(total), indicators };
}

export interface FullRiskAssessment {
  flood: RiskAssessment;
  heatwave: RiskAssessment;
  cyclone: RiskAssessment;
  extremeRainfall: RiskAssessment;
  earthquake: RiskAssessment;
  wildfire: RiskAssessment;
  volcano: RiskAssessment;
  landslide: RiskAssessment;
  drought: RiskAssessment;
  coldWave: RiskAssessment;
  overall: RiskAssessment;
}

export function assessAllRisks(
  weather: WeatherData,
  lat: number,
  lon: number,
  events: DisasterEvent[],
  historical: HistoricalData | null,
  riverDischarge: number | null
): FullRiskAssessment {
  const flood = assessFlood(weather, riverDischarge);
  const heatwave = assessHeatwave(weather, historical);
  const cyclone = assessCyclone(weather);
  const extremeRainfall = assessExtremeRainfall(weather);
  const earthquake = assessEarthquake(lat, lon, events);
  const wildfire = assessWildfire(lat, lon, events, weather);
  const volcano = assessVolcano(lat, lon, events);
  const landslide = assessLandslide(weather);
  const drought = assessDrought(weather, historical);
  const coldWave = assessColdWave(weather);

  const all = [flood, heatwave, cyclone, extremeRainfall, earthquake, wildfire, volcano, landslide, drought, coldWave];
  const overallScore = Math.round(all.reduce((s, a) => Math.max(s, a.score), 0));
  const overall: RiskAssessment = {
    hazard: "flood",
    score: overallScore,
    level: scoreToLevel(overallScore),
    indicators: all
      .filter((a) => a.score >= 25)
      .sort((a, b) => b.score - a.score)
      .slice(0, 3)
      .map((a) => ({
        name: a.hazard.charAt(0).toUpperCase() + a.hazard.slice(1),
        value: `${a.score}/100`,
        contribution: a.score,
        maxContribution: 100,
        description: `Top contributing hazard`,
      })),
  };

  return {
    flood,
    heatwave,
    cyclone,
    extremeRainfall,
    earthquake,
    wildfire,
    volcano,
    landslide,
    drought,
    coldWave,
    overall,
  };
}

const EARTH_RADIUS = 2;
const ROTATION_SPEED = 0.03;

function latLonToVector3(lat: number, lon: number, radius: number): THREE.Vector3 {
  const phi = (90 - lat) * (Math.PI / 180);
  const theta = (lon + 180) * (Math.PI / 180);
  return new THREE.Vector3(
    -(radius * Math.sin(phi) * Math.cos(theta)),
    radius * Math.cos(phi),
    radius * Math.sin(phi) * Math.sin(theta)
  );
}

interface EarthProps {
  onGlobeClick: (lat: number, lon: number) => void;
  isLight: boolean;
}

function Earth({ onGlobeClick, isLight }: EarthProps) {
  const meshRef = useRef<THREE.Mesh>(null);

  const texture = useMemo(() => {
    const loader = new THREE.TextureLoader();
    const url = "https://unpkg.com/three-globe/example/img/earth-blue-marble.jpg";
    return loader.load(url);
  }, [isLight]);

  const handleClick = (e: ThreeEvent<MouseEvent>) => {
    e.stopPropagation();
    const point = e.point.clone();
    const local = e.object.worldToLocal(point);
    const r = local.length();
    const lat = 90 - (Math.acos(local.y / r) * 180) / Math.PI;
    const lon = ((Math.atan2(local.z, -local.x) * 180) / Math.PI) - 180;
    const normalizedLon = ((lon + 540) % 360) - 180;
    onGlobeClick(lat, normalizedLon);
  };

  return (
    <mesh ref={meshRef} onClick={handleClick}>
      <sphereGeometry args={[EARTH_RADIUS, 64, 64]} />
      <meshPhongMaterial map={texture} shininess={10} />
    </mesh>
  );
}

interface MarkerProps {
  lat: number;
  lon: number;
  color: string;
  size: number;
  onClick: () => void;
  pulse?: boolean;
}

function Marker({ lat, lon, color, size, onClick, pulse }: MarkerProps) {
  const ref = useRef<THREE.Mesh>(null);
  const position = useMemo(() => latLonToVector3(lat, lon, EARTH_RADIUS * 1.01), [lat, lon]);

  useFrame((state) => {
    if (ref.current && pulse) {
      const s = size * (1 + Math.sin(state.clock.elapsedTime * 3) * 0.3);
      ref.current.scale.setScalar(s);
    }
  });

  return (
    <mesh ref={ref} position={position} onClick={(e) => { e.stopPropagation(); onClick(); }}>
      <sphereGeometry args={[size, 12, 12]} />
      <meshBasicMaterial color={color} />
    </mesh>
  );
}

interface RingMarkerProps {
  lat: number;
  lon: number;
  color: string;
  size: number;
  onClick: () => void;
}

function RingMarker({ lat, lon, color, size, onClick }: RingMarkerProps) {
  const ref = useRef<THREE.Mesh>(null);
  const position = useMemo(() => latLonToVector3(lat, lon, EARTH_RADIUS * 1.02), [lat, lon]);

  useFrame((state) => {
    if (ref.current) {
      const t = (state.clock.elapsedTime * 1.5) % 1;
      const s = 1 + t * 3;
      ref.current.scale.setScalar(s);
      (ref.current.material as THREE.MeshBasicMaterial).opacity = (1 - t) * 0.6;
    }
  });

  const orientation = useMemo(() => {
    const pos = latLonToVector3(lat, lon, EARTH_RADIUS);
    const quaternion = new THREE.Quaternion();
    quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), pos.normalize());
    return quaternion;
  }, [lat, lon]);

  return (
    <mesh ref={ref} position={position} quaternion={orientation} onClick={(e) => { e.stopPropagation(); onClick(); }}>
      <ringGeometry args={[size * 0.8, size, 24]} />
      <meshBasicMaterial color={color} transparent opacity={0.6} side={THREE.DoubleSide} />
    </mesh>
  );
}

interface RotatingGroupProps {
  children: React.ReactNode;
}

function RotatingGroup({ children }: RotatingGroupProps) {
  const groupRef = useRef<THREE.Group>(null);

  useFrame((_, delta) => {
    if (groupRef.current) {
      groupRef.current.rotation.y += delta * ROTATION_SPEED;
    }
  });

  return <group ref={groupRef}>{children}</group>;
}

interface FlyToControllerProps {
  target: { lat: number; lon: number } | null;
  controlsRef: React.RefObject<{ target: THREE.Vector3 } | null>;
}

function FlyToController({ target, controlsRef }: FlyToControllerProps) {
  const { camera } = useThree();
  const animatingRef = useRef(false);
  const targetRef = useRef(new THREE.Vector3());
  const startRef = useRef(new THREE.Vector3());
  const progressRef = useRef(0);

  useEffect(() => {
    if (!target) return;
    const pos = latLonToVector3(target.lat, target.lon, EARTH_RADIUS * 3.5);
    startRef.current.copy(camera.position);
    targetRef.current.copy(pos);
    progressRef.current = 0;
    animatingRef.current = true;
  }, [target, camera]);

  useFrame((_, delta) => {
    if (!animatingRef.current) return;
    progressRef.current = Math.min(1, progressRef.current + delta * 1.5);
    const t = 1 - Math.pow(1 - progressRef.current, 3);
    camera.position.lerpVectors(startRef.current, targetRef.current, t);
    if (controlsRef.current) {
      controlsRef.current.target.set(0, 0, 0);
    }
    if (progressRef.current >= 1) animatingRef.current = false;
  });

  return null;
}

import { Play, Pause, RotateCcw, SkipBack, SkipForward, Clock } from "lucide-react";
import { useEffect, useRef } from "react";
import type { ScenarioId, TimelineFrame } from "@/types";
import { SCENARIO_LIST } from "@/data/scenarios";
import { SimulatedBadge } from "@/components/ui/AlertBadge";

interface SimulationControlsProps {
  scenarioId: ScenarioId;
  onScenarioChange: (id: ScenarioId) => void;
  currentHour: number;
  onHourChange: (hour: number) => void;
  isPlaying: boolean;
  onPlayPause: () => void;
  timeline: TimelineFrame[];
}

export default function SimulationControls({
  scenarioId,
  onScenarioChange,
  currentHour,
  onHourChange,
  isPlaying,
  onPlayPause,
  timeline,
}: SimulationControlsProps) {
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

  useEffect(() => {
    if (isPlaying && timeline.length > 0) {
      intervalRef.current = setInterval(() => {
        onHourChange(Math.min(currentHour + 1, timeline.length - 1));
      }, 1000);
    }
    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  }, [isPlaying, currentHour, timeline.length, onHourChange]);

  const maxHour = timeline.length > 0 ? timeline.length - 1 : 0;
  const currentFrame = timeline[currentHour];

  return (
    <div className="rounded-xl border border-orange-500/30 bg-slate-900/80 backdrop-blur-xl p-3 space-y-3">
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-2">
          <span className="text-sm font-semibold text-white">Simulation</span>
          <SimulatedBadge />
        </div>
      </div>

      {/* Scenario selector */}
      <div className="flex flex-wrap gap-1.5">
        {SCENARIO_LIST.map((s) => (
          <button
            key={s.id}
            onClick={() => onScenarioChange(s.id)}
            className={`rounded-lg px-2.5 py-1.5 text-xs font-medium transition-colors ${
              scenarioId === s.id
                ? "bg-orange-500/20 text-orange-400 border border-orange-500/40"
                : "bg-white/5 text-gray-400 hover:text-white border border-transparent"
            }`}
            title={s.description}
          >
            {s.name}
          </button>
        ))}
      </div>

      {scenarioId !== "live" && timeline.length > 0 && (
        <>
          {/* Time slider */}
          <div className="space-y-2">
            <div className="flex items-center justify-between text-xs">
              <span className="text-gray-400 flex items-center gap-1">
                <Clock className="h-3 w-3" />
                {currentFrame?.label ?? `T+${currentHour}h`}
              </span>
              <span className="text-gray-600">{currentHour} / {maxHour}h</span>
            </div>
            <div className="flex items-center gap-2">
              <button
                onClick={() => onHourChange(0)}
                className="rounded p-1 text-gray-400 hover:bg-white/5"
                title="Reset"
              >
                <RotateCcw className="h-3.5 w-3.5" />
              </button>
              <button
                onClick={() => onHourChange(Math.max(0, currentHour - 1))}
                className="rounded p-1 text-gray-400 hover:bg-white/5"
              >
                <SkipBack className="h-3.5 w-3.5" />
              </button>
              <button
                onClick={onPlayPause}
                className="rounded-lg bg-orange-500/20 p-1.5 text-orange-400 hover:bg-orange-500/30"
              >
                {isPlaying ? <Pause className="h-4 w-4" /> : <Play className="h-4 w-4" />}
              </button>
              <button
                onClick={() => onHourChange(Math.min(maxHour, currentHour + 1))}
                className="rounded p-1 text-gray-400 hover:bg-white/5"
              >
                <SkipForward className="h-3.5 w-3.5" />
              </button>
              <input
                type="range"
                min={0}
                max={maxHour}
                value={currentHour}
                onChange={(e) => onHourChange(Number(e.target.value))}
                className="flex-1 h-1.5 rounded-full bg-white/10 accent-orange-500 cursor-pointer"
              />
            </div>
          </div>

          {/* Current frame info */}
          {currentFrame && (
            <div className="rounded-lg bg-slate-800/50 p-2 grid grid-cols-3 gap-2 text-[10px]">
              {currentFrame.weather.temperature !== undefined && (
                <FrameStat label="Temp" value={`${currentFrame.weather.temperature.toFixed(0)}°C`} />
              )}
              {currentFrame.weather.windSpeed !== undefined && (
                <FrameStat label="Wind" value={`${currentFrame.weather.windSpeed.toFixed(0)} km/h`} />
              )}
              {currentFrame.weather.rain !== undefined && (
                <FrameStat label="Rain" value={`${currentFrame.weather.rain.toFixed(1)} mm`} />
              )}
              {currentFrame.weather.windGusts !== undefined && (
                <FrameStat label="Gusts" value={`${currentFrame.weather.windGusts.toFixed(0)} km/h`} />
              )}
              {currentFrame.weather.pressure !== undefined && (
                <FrameStat label="Pressure" value={`${currentFrame.weather.pressure.toFixed(0)} hPa`} />
              )}
              {currentFrame.alertLevel && (
                <FrameStat
                  label="Risk"
                  value={currentFrame.alertLevel.toUpperCase()}
                  color={currentFrame.alertLevel === "severe" ? "#ef4444" : currentFrame.alertLevel === "high" ? "#f97316" : currentFrame.alertLevel === "moderate" ? "#eab308" : "#22c55e"}
                />
              )}
            </div>
          )}
        </>
      )}
    </div>
  );
}

function FrameStat({ label, value, color }: { label: string; value: string; color?: string }) {

import { useState, useEffect, useRef } from "react";
import { Search, MapPin, Loader2 } from "lucide-react";
import { searchCity, type GeoResult } from "@/lib/api";

interface CitySearchProps {
  onResult: (result: GeoResult) => void;
}

export default function CitySearch({ onResult }: CitySearchProps) {
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<GeoResult[]>([]);
  const [loading, setLoading] = useState(false);
  const [open, setOpen] = useState(false);
  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    if (timerRef.current) clearTimeout(timerRef.current);
    if (query.length < 2) {
      setResults([]);
      return;
    }
    setLoading(true);
    timerRef.current = setTimeout(async () => {
      const res = await searchCity(query);
      setResults(res);
      setLoading(false);
      setOpen(true);
    }, 400);
    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
  }, [query]);

  const handleSelect = (r: GeoResult) => {
    onResult(r);
    setQuery(`${r.name}, ${r.country}`);
    setOpen(false);
  };

  return (
    <div className="relative w-full">
      <div className="relative">
        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
        <input
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          onFocus={() => results.length > 0 && setOpen(true)}
          onBlur={() => setTimeout(() => setOpen(false), 200)}
          placeholder="Search city..."
          className="w-full rounded-lg border border-white/10 bg-slate-900/80 pl-9 pr-3 py-2 text-sm text-white placeholder-gray-500 outline-none focus:border-blue-500/50"
        />
        {loading && <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 animate-spin text-gray-400" />}
      </div>
      {open && results.length > 0 && (
        <div className="absolute z-50 mt-1 w-full rounded-lg border border-white/10 bg-slate-900/95 backdrop-blur-xl shadow-2xl max-h-60 overflow-y-auto">
          {results.map((r, i) => (
            <button
              key={`${r.name}-${i}`}
              onClick={() => handleSelect(r)}
              className="flex w-full items-center gap-2 px-3 py-2 text-left text-sm text-gray-200 hover:bg-white/5"
            >
              <MapPin className="h-3.5 w-3.5 text-blue-400 flex-shrink-0" />
              <span className="font-medium">{r.name}</span>
              <span className="text-gray-500">{r.country}{r.admin1 ? `, ${r.admin1}` : ""}</span>
            </button>
          ))}
        </div>
      )}
    </div>
  );
}

import { useMemo } from "react";
import type { WeatherData } from "@/types";

interface ForecastChartProps {
  weather: WeatherData;
  type: "temperature" | "rain" | "wind";
}

export default function ForecastChart({ weather, type }: ForecastChartProps) {
  const data = useMemo(() => {
    if (type === "temperature") {
      return {
        values: weather.daily.temperatureMax.slice(0, 7),
        mins: weather.daily.temperatureMin.slice(0, 7),
        label: "Temperature (°C)",
        color: "#f59e0b",
        minColor: "#3b82f6",
      };
    }
    if (type === "rain") {
      return {
        values: weather.daily.precipitationSum.slice(0, 7),
        mins: undefined,
        label: "Rain (mm)",
        color: "#06b6d4",
        minColor: undefined,
      };
    }
    return {
      values: weather.daily.windGustsMax.slice(0, 7),
      mins: undefined,
      label: "Wind Gusts (km/h)",
      color: "#10b981",
      minColor: undefined,
    };
  }, [weather, type]);

  const max = Math.max(...data.values, ...(data.mins ?? [0]), 1) * 1.2;
  const width = 320;
  const height = 120;
  const padding = 30;
  const chartW = width - padding * 2;
  const chartH = height - padding;
  const barW = chartW / 7 - 6;

  return (
    <div className="w-full">
      <div className="mb-2 text-xs font-medium text-gray-400">{data.label} — 7-day forecast</div>
      <svg width="100%" viewBox={`0 0 ${width} ${height}`} preserveAspectRatio="xMidYMid meet">
        {[0, 0.25, 0.5, 0.75, 1].map((t) => (
          <line
            key={t}
            x1={padding}
            x2={width - padding}
            y1={padding + chartH * (1 - t)}
            y2={padding + chartH * (1 - t)}
            stroke="currentColor"
            strokeWidth="0.5"
            className="text-white/10"
          />
        ))}
        {data.values.map((v, i) => {
          const h = (v / max) * chartH;
          const x = padding + i * (barW + 6);
          const y = padding + chartH - h;
          const date = new Date(weather.daily.time?.[i] ?? weather.daily.time?.[0] ?? new Date().toISOString());
          const dayLabel = date.toLocaleDateString("en", { weekday: "short" });
          return (
            <g key={i}>
              <rect x={x} y={y} width={barW} height={h} fill={data.color} rx={3} opacity={0.8} />
              {data.mins && (
                <rect
                  x={x}
                  y={padding + chartH - (data.mins[i] / max) * chartH}
                  width={barW}
                  height={(data.mins[i] / max) * chartH}
                  fill={data.minColor}
                  rx={3}
                  opacity={0.5}
                />
              )}
              <text x={x + barW / 2} y={height - 4} textAnchor="middle" fontSize="8" fill="currentColor" className="text-gray-500">
                {dayLabel}
              </text>
              <text x={x + barW / 2} y={y - 4} textAnchor="middle" fontSize="8" fill={data.color} fontWeight="bold">
                {v.toFixed(type === "rain" ? 0 : 0)}
              </text>
            </g>
          );
        })}
      </svg>
    </div>
  );
}

export default function RiskGauge({ score, level, size = 120, label }: RiskGaugeProps) {
  const meta = RISK_LEVEL_META[level];
  const radius = (size - 16) / 2;
  const circumference = 2 * Math.PI * radius;
  const offset = circumference - (score / 100) * circumference;

  return (
    <div className="flex flex-col items-center gap-1">
      <div className="relative" style={{ width: size, height: size }}>
        <svg width={size} height={size} className="transform -rotate-90">

            r={radius}

            stroke="currentColor"
            strokeWidth="8"
            className="text-white/10"
          />
          <circle
            cx={size / 2}
            cy={size / 2}
            r={radius}
            fill="none"
            stroke={meta.color}
            strokeWidth="8"
            strokeDasharray={circumference}
            strokeDashoffset={offset}
            strokeLinecap="round"
            style={{ transition: "stroke-dashoffset 0.8s ease" }}
          />
        </svg>
        <div className="absolute inset-0 flex flex-col items-center justify-center">
